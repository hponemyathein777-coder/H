<script>
  function calculate() {
    const height = parseFloat(document.getElementById('height').value) || 0;
    const hollow = parseFloat(document.getElementById('hollow').value) || 0;
    const longGap = parseFloat(document.getElementById('longGap').value) || 0;
    const shortGap = parseFloat(document.getElementById('shortGap').value) || 0;
    const strips = parseFloat(document.getElementById('strips').value) || 0;

    if (strips > 0) {
      const remainingSpace = height - (hollow * strips) - longGap - shortGap;
      const gapResult = remainingSpace / (strips - 1);
      document.getElementById('gap').value = gapResult.toFixed(2);
    } else {
      document.getElementById('gap').value = "0";
    }
  }

  function changeStrips(val) {
    const stripsInput = document.getElementById('strips');
    let currentVal = parseInt(stripsInput.value) || 0;
    currentVal += val;
    if (currentVal < 1) currentVal = 1;
    stripsInput.value = currentVal;
    calculate();
  }

  calculate();
</script>
